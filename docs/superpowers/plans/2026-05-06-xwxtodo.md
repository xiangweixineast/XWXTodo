# XWXTodo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建 XWXTodo macOS 14+ 第一版：一个纯本地 TODO 应用，主屏幕顶部常驻黑色刘海屏，悬停后展开为可管理 TODO 的顶部弹窗。

**Architecture:** 使用现有 `XWXTodo/XWXTodo.xcodeproj` 作为工程基础。AppKit 负责系统级顶部覆盖层窗口，SwiftUI 负责刘海屏和弹窗内容，`TodoStore` 负责业务规则，`SQLiteTodoRepository` 负责本地 SQLite 持久化。数据/业务层按 TDD 实施；桌面客户端 UI 不做 TDD，以构建、单元测试和手动验收验证。

**Tech Stack:** Swift 5、SwiftUI、AppKit、XCTest、SQLite3、Xcode 16 工程、macOS 14+。

- `CompletedListView.swift`：只读已完成列表。

`InMemoryTodoRepository.swift`：

```swift
import Foundation
@testable import XWXTodo

final class InMemoryTodoRepository: TodoRepository {
    private var items: [TodoItem]

    init(items: [TodoItem] = []) {
        self.items = items
    }

    func loadAll() throws -> [TodoItem] {
        items
    }

    func insert(_ item: TodoItem) throws {
        items.append(item)
    }

    func update(_ item: TodoItem) throws {
        guard let index = items.firstIndex(where: { $0.id == item.id }) else { return }
        items[index] = item
    }

    func delete(id: UUID) throws {
        items.removeAll { $0.id == id && $0.status != .completed }
    }

    func setDoing(id: UUID, updatedAt: Date) throws {
        for index in items.indices {
            if items[index].status == .doing {
                items[index].status = .pending
                items[index].updatedAt = updatedAt
            }
        }
        guard let targetIndex = items.firstIndex(where: { $0.id == id && $0.status != .completed }) else { return }
        items[targetIndex].status = .doing
        items[targetIndex].updatedAt = updatedAt
    }

    func complete(id: UUID, completedAt: Date) throws {
        guard let index = items.firstIndex(where: { $0.id == id }) else { return }
        items[index].status = .completed
        items[index].completedAt = completedAt
        items[index].updatedAt = completedAt
    }
}
```

- [ ] **Step 5: 实现 `TodoStore`**

`TodoStore.swift`：

```swift
import Foundation
import Combine

@MainActor
final class TodoStore: ObservableObject {
    @Published private(set) var todos: [TodoItem] = []
    @Published private(set) var errorMessage: String?

    private let repository: TodoRepository
    private let now: () -> Date

    init(repository: TodoRepository, now: @escaping () -> Date = Date.init) throws {
        self.repository = repository
        self.now = now
        try reload()
    }

    var activeTodos: [TodoItem] {
        todos
            .filter { $0.status == .pending || $0.status == .doing }
            .sorted { lhs, rhs in
                if lhs.sortOrder == rhs.sortOrder { return lhs.createdAt < rhs.createdAt }
                return lhs.sortOrder < rhs.sortOrder
            }
    }

    var completedTodos: [TodoItem] {
        todos
            .filter { $0.status == .completed }
            .sorted {
                ($0.completedAt ?? .distantPast) > ($1.completedAt ?? .distantPast)
            }
    }

    var doingTodo: TodoItem? {
        todos.first { $0.status == .doing }
    }

    var notchTitle: String {
        doingTodo?.title ?? "XWXTodo"
    }

    func reload() throws {
        todos = try repository.loadAll()
        errorMessage = nil
    }

    func addTodo(title: String) throws {
        let trimmed = title.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else { return }
        let timestamp = now()
        let nextSortOrder = ((todos.map(\.sortOrder).max()) ?? -1) + 1
        let item = TodoItem(
            id: UUID(),
            title: trimmed,
            status: .pending,
            createdAt: timestamp,
            updatedAt: timestamp,
            completedAt: nil,
            sortOrder: nextSortOrder
        )
        try repository.insert(item)
        try reload()
    }

    func editTodo(id: UUID, title: String) throws {
        let trimmed = title.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else { return }
        guard var item = todos.first(where: { $0.id == id && $0.status != .completed }) else { return }
        item.title = trimmed
        item.updatedAt = now()
        try repository.update(item)
        try reload()
    }

    func deleteTodo(id: UUID) throws {
        guard todos.contains(where: { $0.id == id && $0.status != .completed }) else { return }
        try repository.delete(id: id)
        try reload()
    }

    func startTodo(id: UUID) throws {
        try repository.setDoing(id: id, updatedAt: now())
        try reload()
    }

    func completeTodo(id: UUID) throws {
        try repository.complete(id: id, completedAt: now())
        try reload()
    }
}
```

```swift
    private func makeTemporaryDatabaseURL() -> URL {
        FileManager.default.temporaryDirectory
            .appendingPathComponent(UUID().uuidString)
            .appendingPathExtension("sqlite")

        XCTAssertEqual(try repository.loadAll(), [])

    func testPersistsInsertedTodoAcrossRepositoryInstances() throws {
        let url = makeTemporaryDatabaseURL()
        let createdAt = Date(timeIntervalSince1970: 1_800_000_000)
        let item = TodoItem(id: UUID(), title: "写代码", status: .pending, createdAt: createdAt, updatedAt: createdAt, completedAt: nil, sortOrder: 0)

        XCTAssertEqual(reloaded, [item])
    }

    func testSetDoingIsTransactionalAndLeavesOnlyOneDoing() throws {
        let repository = try SQLiteTodoRepository(databaseURL: makeTemporaryDatabaseURL())

        let first = TodoItem(id: UUID(), title: "A", status: .pending, createdAt: date, updatedAt: date, completedAt: nil, sortOrder: 0)

        try repository.insert(second)
```

`SQLiteError.swift`：

```swift
import Foundation

enum SQLiteError: Error, LocalizedError {
    case openFailed(String)
    case prepareFailed(String)
    case stepFailed(String)
    case bindFailed(String)
    case migrationFailed(String)

    var errorDescription: String? {
        switch self {
        case .openFailed(let message): return "数据库打开失败：\(message)"
        case .prepareFailed(let message): return "SQL 准备失败：\(message)"
        case .stepFailed(let message): return "SQL 执行失败：\(message)"
        case .bindFailed(let message): return "SQL 参数绑定失败：\(message)"
        case .migrationFailed(let message): return "数据库迁移失败：\(message)"
        }
    }
}
```

- [ ] **Step 4: 实现 SQLite repository**

`SQLiteTodoRepository.swift` 使用 `import SQLite3`，初始化时创建父目录和 schema：

```sql
CREATE TABLE IF NOT EXISTS todos (
  id TEXT PRIMARY KEY NOT NULL,
  title TEXT NOT NULL,
  status TEXT NOT NULL,
  created_at REAL NOT NULL,
  updated_at REAL NOT NULL,
  completed_at REAL,
  sort_order INTEGER NOT NULL
);
CREATE UNIQUE INDEX IF NOT EXISTS idx_todos_single_doing
ON todos(status)
WHERE status = 'doing';
```

关键实现要求：

- `databaseURL` init 用于测试。
- `init() throws` 使用 `defaultDatabaseURL()`。
- `defaultDatabaseURL()` 返回 `~/Library/Application Support/XWXTodo/xwxtodo.sqlite`。
- `loadAll()` 按 `sort_order ASC, created_at ASC` 读取。
- `insert(_:)`、`update(_:)` 使用 prepared statement，不拼接用户输入。
- `setDoing(id:updatedAt:)` 在 `BEGIN IMMEDIATE` / `COMMIT` 中执行：
  - `UPDATE todos SET status='pending', updated_at=? WHERE status='doing';`
  - `UPDATE todos SET status='doing', updated_at=? WHERE id=? AND status <> 'completed';`
- `complete(id:completedAt:)` 写 `status='completed'`、`completed_at`、`updated_at`。
- `delete(id:)` 只删除 `status <> 'completed'` 的记录。
- 所有 statement 都要 `sqlite3_finalize`。

```bash
echo "$DIST_DIR/XWXTodo.zip"
```
